{#- Resolves the active menu item, then renders the menu -#} {%- set_global current_item = false -%} {%- set_global base_item = false -%} {%- if config.extra.menu_items %} {%- set menu_items = config.extra.menu_items -%} {%- for item in menu_items %} {%- set abs_item_url = item.url | replace(from="$BASE_URL", to=config.base_url) -%} {%- set is_current = current_url == abs_item_url ~ "/" or current_url is starting_with(pat=abs_item_url) -%} {%- set is_base = abs_item_url == config.base_url or abs_item_url == config.base_url ~ "/" -%} {%- if is_base %} {%- set_global base_item = item -%} {% endif -%} {%- if is_current and not is_base %} {%- set_global current_item = item -%} {% endif -%} {% endfor -%} {%- if not current_item and base_item %} {# Did not match any menu URLs -- assume it's a blog post #} {%- set_global current_item = base_item -%} {% endif -%} {% include "partials/menu_for.html" %} {% endif -%}
